cmake_minimum_required(VERSION 3.22)
project(DynarmicArm64Benchmark LANGUAGES C CXX ASM)

set(DYNARMIC_SOURCE_DIR "" CACHE PATH "Unmodified Dynarmic checkout")
if(NOT EXISTS "${DYNARMIC_SOURCE_DIR}/src/dynarmic/interface/A64/a64.h")
    message(FATAL_ERROR "Set DYNARMIC_SOURCE_DIR to a Dynarmic checkout")
endif()
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_FRONTENDS A64 CACHE STRING "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DYNARMIC_SOURCE_DIR}" dynarmic)
add_executable(dynarmic_arm64_compare_bench ../arm64_compare.cpp)
target_compile_features(dynarmic_arm64_compare_bench PRIVATE cxx_std_20)
target_compile_definitions(dynarmic_arm64_compare_bench PRIVATE SVM_BENCH_DYNARMIC=1)
target_link_libraries(dynarmic_arm64_compare_bench PRIVATE dynarmic)
