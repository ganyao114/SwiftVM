foreach(guest IN ITEMS x86 arm64 arm32)
    if(guest STREQUAL "x86")
        set(guest_id 1)
    elseif(guest STREQUAL "arm64")
        set(guest_id 2)
    else()
        set(guest_id 3)
    endif()
    add_executable(svm_fuzz_decode_${guest} decode.cpp)
    target_compile_definitions(svm_fuzz_decode_${guest} PRIVATE SVM_FUZZ_ISA=${guest_id})
    target_link_libraries(svm_fuzz_decode_${guest} PRIVATE svm_correctness_core)
    target_link_options(svm_fuzz_decode_${guest} PRIVATE -fsanitize=fuzzer,address,undefined)
    set_target_properties(svm_fuzz_decode_${guest} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
